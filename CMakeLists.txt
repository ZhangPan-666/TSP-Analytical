cmake_minimum_required(VERSION 3.20)
project(TSP_Analytical LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(MSVC)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
  add_compile_options(/arch:AVX2 /fp:precise)
  if(CMAKE_CXX_COMPILER_ID MATCHES "IntelLLVM|Clang")
    add_compile_options(/clang:-mfma /clang:-ffp-contract=off)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    add_compile_options(/Zc:lambda)
  endif()
else()
  add_compile_options(-fno-fast-math -ffp-contract=off)
endif()
foreach(name Work1_GravityEstimate Work1_MagneticEstimate)
  add_executable(${name} ${name}.cpp)
endforeach()
foreach(name VerifyTSP VerifyScalar VerifyRegular)
  add_executable(${name} tests/${name}.cpp)
endforeach()
enable_testing()
add_test(NAME TSP_Numerical_Validation COMMAND VerifyTSP)
add_test(NAME TSP_Scalar_Validation COMMAND VerifyScalar)
add_test(NAME TSP_Regular_Validation COMMAND VerifyRegular)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  foreach(name Work1_GravityEstimateOpenMP Work1_MagneticEstimateOpenMP)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
  endforeach()
endif()
find_package(MPI QUIET COMPONENTS CXX)
if(NOT MPI_CXX_FOUND AND MSVC)
  find_path(TSP_MPI_INCLUDE mpi.h HINTS "$ENV{I_MPI_ROOT}/include" "C:/Program Files (x86)/Intel/oneAPI/mpi/latest/include")
  find_library(TSP_MPI_LIBRARY impi HINTS "$ENV{I_MPI_ROOT}/lib" "C:/Program Files (x86)/Intel/oneAPI/mpi/latest/lib")
endif()
if(MPI_CXX_FOUND OR (TSP_MPI_INCLUDE AND TSP_MPI_LIBRARY))
  foreach(name Work1_GravityEstimateMPI Work1_MagneticEstimateMPI Work2_GravityEstimateVzMPI Work2_MagneticEstimateVectorMPI)
    add_executable(${name} ${name}.cpp)
    if(MPI_CXX_FOUND)
      target_link_libraries(${name} PRIVATE MPI::MPI_CXX)
    else()
      target_include_directories(${name} PRIVATE ${TSP_MPI_INCLUDE})
      target_link_libraries(${name} PRIVATE ${TSP_MPI_LIBRARY})
    endif()
  endforeach()
endif()
get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(name IN LISTS targets)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endforeach()
